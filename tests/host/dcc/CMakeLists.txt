cmake_minimum_required(VERSION 3.20)
project(hexwallet_dcc_codec LANGUAGES CXX)
get_filename_component(WALLET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(dcc_test dcc_test.cpp "${WALLET_ROOT}/DccChainCodec.cpp")
target_include_directories(dcc_test PRIVATE "${WALLET_ROOT}")
target_compile_features(dcc_test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(dcc_test PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(dcc_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME dcc_exact_projection COMMAND dcc_test)
