cmake_minimum_required(VERSION 3.20)
project(hexwallet_ui_regression LANGUAGES C CXX)
get_filename_component(WALLET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(LVGL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../lvgl_real" CACHE PATH "LVGL 9.5 source checkout")
set(LV_BUILD_CONF_PATH "${WALLET_ROOT}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
add_executable(ui_test ui_test.cpp "${WALLET_ROOT}/Mascot.cpp"
  "${WALLET_ROOT}/MaidSprite.cpp" "${WALLET_ROOT}/QrCode.cpp"
  "${WALLET_ROOT}/CoinIcons.cpp")
target_compile_features(ui_test PRIVATE cxx_std_17)
target_compile_definitions(ui_test PRIVATE HEXWALLET_ENABLE_LVGL=1 HEXWALLET_BOARD=0)
target_include_directories(ui_test BEFORE PRIVATE "${LVGL_SOURCE_DIR}/src")
target_include_directories(ui_test PRIVATE "${WALLET_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/../mock")
target_link_libraries(ui_test PRIVATE lvgl::lvgl)
if(MSVC)
  target_compile_options(ui_test PRIVATE /utf-8 /W3)
endif()
