cmake_minimum_required(VERSION 3.20)
project(hexwallet_ui_regression LANGUAGES C CXX)
get_filename_component(WALLET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(LVGL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../lvgl_real" CACHE PATH "LVGL 9.5 source checkout")
set(LV_BUILD_CONF_PATH "${WALLET_ROOT}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
add_executable(ui_test ui_test.cpp "${WALLET_ROOT}/Mascot.cpp"
  "${WALLET_ROOT}/MaidSprite.cpp" "${WALLET_ROOT}/QrCode.cpp"
  "${WALLET_ROOT}/CoinIcons.cpp")
target_compile_features(ui_test PRIVATE cxx_std_17)
target_compile_definitions(ui_test PRIVATE HEXWALLET_ENABLE_LVGL=1 HEXWALLET_BOARD=0)
target_include_directories(ui_test BEFORE PRIVATE "${LVGL_SOURCE_DIR}/src")
target_include_directories(ui_test PRIVATE "${WALLET_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/../mock")
target_link_libraries(ui_test PRIVATE lvgl::lvgl)
if(MSVC)
  target_compile_options(ui_test PRIVATE /utf-8 /W3)
endif()

foreach(profile IN ITEMS p4_eink s3_display)
  if(profile STREQUAL "p4_eink")
    set(board_id 7)
  else()
    set(board_id 1)
  endif()
  add_executable(board_${profile}_test board_runtime_test.cpp
    "${WALLET_ROOT}/EinkPanel.cpp" "${WALLET_ROOT}/EinkFrame.cpp")
  target_compile_features(board_${profile}_test PRIVATE cxx_std_17)
  target_compile_definitions(board_${profile}_test PRIVATE
    HEXWALLET_ENABLE_LVGL=1 HEXWALLET_BOARD=${board_id} HEXWALLET_BOARD_RUNTIME_TEST=1)
  target_include_directories(board_${profile}_test BEFORE PRIVATE "${LVGL_SOURCE_DIR}/src")
  target_include_directories(board_${profile}_test PRIVATE
    "${WALLET_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/../mock")
  target_link_libraries(board_${profile}_test PRIVATE lvgl::lvgl)
  if(MSVC)
    target_compile_options(board_${profile}_test PRIVATE /utf-8 /W4)
  endif()
endforeach()
enable_testing()
add_test(NAME ui COMMAND ui_test)
add_test(NAME board_p4_eink COMMAND board_p4_eink_test)
add_test(NAME board_s3_display COMMAND board_s3_display_test)
set_tests_properties(ui board_p4_eink board_s3_display PROPERTIES TIMEOUT 30)
